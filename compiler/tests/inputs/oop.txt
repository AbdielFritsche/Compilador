class WaterSensor {
    private var sensor_id = 404;
    public var is_active = !!1;
    public var water_level = 0.0;

    public void calibrate(float offset) {
        this.water_level += offset;
        
        if (this.water_level >= 100.0) {
            print("Alerta: Nivel critico");
        } else {
            print("Nivel estable");
        }
    }

    public float getLevel() {
        return this.water_level;
    }
}

program main {
    # Instanciacion de clase
    var main_sensor = new WaterSensor();
    var readings = 0;

    # Bucle de simulacion con operadores logicos y de comparacion
    while (main_sensor.getLevel() < 50.0 && readings !== 10) {
        main_sensor.calibrate(12.5);
        readings++;
    }

    # Operadores de identidad estricta y corrimiento de bits
